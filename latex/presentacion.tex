% --- DIAPOSITIVA 4: MARCO TEÓRICO - POLINOMIOS ORTOGONALES Y CEROS ---
\begin{frame}{Marco Teórico: Polinomios Ortogonales y Ceros}
	\begin{block}{Conceptos Fundamentales} \footnotesize % Tamaño de fuente reducido
		\begin{itemize}
			\item \textbf{Espacio de polinomios} $\mathbb{P}[z]$ y subespacios $\mathbb{P}_n[z]$.
			\item \textbf{Producto interno} $\langle p,q\rangle = \int p(z)\overline{q(z)} d\mu(z)$.
			\item \textbf{Polinomios ortonormales} $\{\varphi_n\}_{n\ge0}$: Gram-Schmidt a $\{1,z,z^2,\dots\}$, $\langle \varphi_n, \varphi_k \rangle = \delta_{nk}$.
			\item \textbf{Ceros de un polinomio} $Z(p) = \{z \in \mathbb{C} : p(z)=0\}$.
			\item \textbf{Conjunto total de ceros} $Z = \bigcup_{n=0}^\infty Z(\varphi_n)$.
		\end{itemize}
	\end{block}
	\pause
	\begin{alertblock}{Matrices de Momentos} \footnotesize % Tamaño de fuente reducido
		Dada una base $\{z^k\}_{k\ge0}$ y un producto interno $\langle \cdot, \cdot \rangle$:
		\begin{itemize}
			\item Matriz de momentos: $M = (c_{i,j})_{i,j\ge0}$ con $c_{i,j} = \langle z^i, z^j \rangle$.
			\item Si $\langle p,q \rangle = \int p \bar{q} d\mu$, entonces $c_{i,j} = \int z^i \bar{z}^j d\mu$.
			\item $M$ es Hermitiana y definida positiva (HPD) \cite{Horn_Johnson_2012}.
			\item Secciones principales: $M_n = (c_{i,j})_{0 \le i,j \le n-1}$.
		\end{itemize}
	\end{alertblock}
\end{frame}

% --- DIAPOSITIVA 5: MARCO TEÓRICO - OPERADOR MULTIPLICACIÓN (I) ---
\begin{frame}{Marco Teórico: Operador Multiplicación $\D(p)=zp$ (I)}
	\begin{block}{Definición y Acotación} \small
		\begin{itemize}
			\item Operador lineal $\D: \mathbb{P}[z] \to \mathbb{P}[z]$ dado por $\D(p(z)) = z \cdot p(z)$.
			\item $\D$ es \textbf{acotado} si existe $C>0$ tal que $\|\D(p)\| \le C \|p\|$ para todo $p \in \mathbb{P}[z]$.
			\item La menor $C$ es la norma del operador $\|\D\|$.
		\end{itemize}
	\end{block}
	\pause
	\begin{alertblock}{Representación Matricial $\mathbf{D}$} \small
		\begin{itemize}
			\item En la base de polinomios ortonormales $\{\varphi_n\}$, $z\varphi_j(z) = \sum_{i=0}^{j+1} d_{i,j} \varphi_i(z)$.
			\item Esto define una matriz infinita $\mathbf{D} = (d_{i,j})_{i,j \ge 0}$.
			\item $\mathbf{D}$ es una matriz de \textbf{Hessenberg superior}: $d_{i,j}=0$ si $i > j+1$.
			      \[
				      \mathbf{D}=\begin{pmatrix}
					      d_{0,0} & d_{0,1} & d_{0,2} & \cdots \\
					      d_{1,0} & d_{1,1} & d_{1,2} & \cdots \\
					      0       & d_{2,1} & d_{2,2} & \cdots \\
					      \vdots  & \vdots  & \ddots  & \ddots
				      \end{pmatrix}
			      \]
		\end{itemize}
	\end{alertblock}
\end{frame}

% --- DIAPOSITIVA 6: MARCO TEÓRICO - OPERADOR MULTIPLICACIÓN (II) - CONEXIÓN ESPECTRAL ---
\begin{frame}{Marco Teórico: Operador Multiplicación (II) - Conexión Espectral}
	Sea $\mathbf{D}_n$ la sección principal de orden $n+1$ de $\mathbf{D}$.
	\begin{alertblock}{Resultado Fundamental}
		\centering
		\Large ¡Los ceros de $\varphi_{n+1}$ son los autovalores de $\mathbf{D}_n^t$!
		$$ Z(\varphi_{n+1}) = \sigma(\mathbf{D}_n^t) $$
		\normalsize (donde $\sigma(A)$ denota el espectro de $A$).
	\end{alertblock}
\end{frame}

% --- DIAPOSITIVA 6A: MARCO TEÓRICO - OPERADOR MULTIPLICACIÓN (II) - CONEXIÓN ESPECTRAL (RESULTADO) ---
\begin{frame}{Marco Teórico: Operador Multiplicación (II) - Conexión Espectral (Resultado)}
	Sea $\mathbf{D}_n$ la sección principal de orden $n+1$ de $\mathbf{D}$.
	\begin{alertblock}{Resultado Fundamental}
		\centering
		\Large ¡Los ceros de $\varphi_{n+1}$ son los autovalores de $\mathbf{D}_n^t$!
		$$ Z(\varphi_{n+1}) = \sigma(\mathbf{D}_n^t) $$
		\normalsize (donde $\sigma(A)$ denota el espectro de $A$).
	\end{alertblock}
\end{frame}

% --- DIAPOSITIVA 6B: MARCO TEÓRICO - OPERADOR MULTIPLICACIÓN (II) - CONEXIÓN ESPECTRAL (DEMOSTRACIÓN) ---
\begin{frame}{Marco Teórico: Operador Multiplicación (II) - Conexión Espectral (Demostración)}
	\begin{block}{Demostración (Idea)} \small
		Si $\varphi_{n+1}(z_0)=0$:
		\begin{align*} z_0\varphi_k(z_0) &= \sum_{i=0}^{k+1} d_{i,k} \varphi_i(z_0) \quad \text{para } k < n \\ z_0\varphi_n(z_0) &= \sum_{i=0}^{n} d_{i,n} \varphi_i(z_0) \quad (\text{ya que } d_{n+1,n}\varphi_{n+1}(z_0)=0) \end{align*}
		Esto se escribe como $z_0 \vec{\Phi}(z_0) = \mathbf{D}_n^t \vec{\Phi}(z_0)$, donde $\vec{\Phi}(z_0) = (\varphi_0(z_0), \dots, \varphi_n(z_0))^t$.
	\end{block}
	\pause \small
	\begin{itemize}
		\item En el caso clásico (medida real), $\mathbf{D}$ es tridiagonal y simétrica (matriz de Jacobi).
		\item Esta conexión permite usar herramientas de álgebra lineal para estudiar los ceros.
	\end{itemize}
\end{frame}

% --- DIAPOSITIVA 7: MARCO TEÓRICO - PARÁMETROS ESPECTRALES CLAVE ---
\begin{frame}{Marco Teórico: Parámetros Espectrales Clave}
	\footnotesize % Reducido tamaño de fuente
	Para una matriz $\mathbf{A}$ (en nuestro caso, $\mathbf{D}_n$):
	\begin{itemize}
		\item \textbf{Radio Espectral $\rho(\mathbf{A})$} \cite{dunford1963linear}: Máximo de los módulos de sus autovalores.
		      $$ \rho(\mathbf{A}) = \max \{ |\lambda| : \lambda \in \sigma(\mathbf{A}) \} $$
		      Nos da la localización del cero $\varphi_{n+1}$ más alejado del origen.
		      \pause
		\item \textbf{Valores Singulares $\sigma_k(\mathbf{A})$} \cite{Horn_Johnson_2012}: Raíces cuadradas de los autovalores de $\mathbf{A}^*\mathbf{A}$ (o $\mathbf{A}^t\mathbf{A}$ si real).
		      $$ \sigma_1 \ge \sigma_2 \ge \dots \ge \sigma_{n+1} \ge 0 $$
		      \begin{itemize}
			      \item $\sigma_1(\mathbf{A}) = \|\mathbf{A}\|$ (norma espectral). Mide si el operador es "acotado" o "explota".
			      \item $\sigma_2(\mathbf{A})$: El segundo valor singular más grande. Veremos que es crucial.
		      \end{itemize}
	\end{itemize}
\end{frame}

		      $$ \sigma_1 \ge \sigma_2 \ge \dots \ge \sigma_{n+1} \ge 0 $$

% --- DIAPOSITIVA 7B: MARCO TEÓRICO - PARÁMETROS ESPECTRALES CLAVE (CONVERGENCIA Y NOTACIÓN) ---
\begin{frame}{Marco Teórico: Parámetros Espectrales Clave (Convergencia y Notación)}
	\footnotesize % Reducido tamaño de fuente
	\begin{block}{Convergencia}
		Si $\mathcal{D}$ es el operador multiplicación (infinito) y $\mathbf{D}_n$ sus secciones:
		$$ \lim_{n\to\infty} \sigma_{1}(\mathbf{D}_n) = \lim_{n\to\infty} \|\mathbf{D}_n\| = \|\mathcal{D}\| $$
		(puede ser $\infty$ si $\mathcal{D}$ no es acotado).
	\end{block}
	\pause
	Usaremos la notación: $\rho_n = \rho(\mathbf{D}_n)$, $\sigma_{1,n} = \sigma_1(\mathbf{D}_n)$, $\sigma_{2,n} = \sigma_2(\mathbf{D}_n)$.
\end{frame}

% --- DIAPOSITIVA 8A: MARCO TEÓRICO - PRODUCTOS INTERNOS DE SOBOLEV (I) - DEFINICIÓN ---
\begin{frame}{Marco Teórico: Productos Internos de Sobolev (I) - Definición}
	\begin{alertblock}{Definición General \cite{escribano2025boundedness}}
		Dados $\boldsymbol{\mu}=(\mu_1, \mu_2)$ medidas de Borel positivas (soporte de $\mu_1$ infinito):
		$$ \langle p,q \rangle_S = \int_{\mathbb{C}} p(z)\overline{q(z)}d\mu_1(z) + \int_{\mathbb{C}} p'(z)\overline{q'(z)}d\mu_2(z) $$
	\end{alertblock}
\end{frame}

% --- DIAPOSITIVA 8B: MARCO TEÓRICO - PRODUCTOS INTERNOS DE SOBOLEV (I) - EJEMPLOS ---
\begin{frame}{Marco Teórico: Productos Internos de Sobolev (I) - Ejemplos}
	\begin{block}{Ejemplos Comunes} \scriptsize % Tamaño de fuente reducido
		\begin{enumerate}
			\item \textbf{Continuo Real:} $\int_a^b p(t)q(t)w_1(t)dt + \int_c^d p'(t)q'(t)w_2(t)dt$
			\item \textbf{Discreto Real:} $\int_a^b p(t)q(t)w_1(t)dt + p'(c)q'(c)$
			\item \textbf{Continuo Complejo (Circunferencias):}
			      $$ \int_{S_{r_b}(b)} p\bar{q}\,dm_{b;r_b} + \int_{S_{r_a}(a)} p'\bar{q}'\,dm_{a;r_a} $$
			      donde $dm_{c;r}$ es la medida de Lebesgue normalizada en $S_r(c)$.
			\item \textbf{Discreto Complejo (Circunferencia y Átomo):}
			      $$ \int_{S_{r_b}(b)} p\bar{q}\,dm_{b;r_b} + p'(c)\overline{q'(c)} $$
		\end{enumerate}
	\end{block}
	\pause \small
	\begin{itemize}
		\item Para términos discretos, $\mu_2$ puede ser una \textbf{medida de Dirac} $\delta_c$, donde $\int f d\delta_c = f(c)$.
	\end{itemize}
\end{frame}

% --- DIAPOSITIVA 9A: MARCO TEÓRICO - PRODUCTOS INTERNOS DE SOBOLEV (IIA) ---
\begin{frame}{Marco Teórico: Productos Sobolev (IIa) - Matriz $\mathbf{M}_S$}
	\begin{block}{Radio del Soporte Combinado} \small
		Se define $R = \max \{ |z| : z \in \operatorname{supp}(\mu_1) \cup \operatorname{supp}(\mu_2) \}$.
		\begin{itemize}
			\item Caso clásico (una medida $\mu$): $R = \max \{|z| : z \in \operatorname{supp}(\mu)\}$. Se sabe que $\|\mathcal{D}\| = R$ si $\operatorname{supp}(\mu)$ es acotado.
		\end{itemize}
	\end{block}
	\pause
	\begin{alertblock}{Matriz de Momentos de Sobolev $\mathbf{M}_S$ \cite{escribano2025boundedness}} \small
		Sean $\mathbf{M}_1 = \mathbf{M}(\mu_1)$ y $\mathbf{M}_2 = \mathbf{M}(\mu_2)$ las matrices de momentos.
		La matriz de momentos de Sobolev es:
		$$ \mathbf{M}_S = \mathbf{M}_1 + \boldsymbol{\Lambda} \mathbf{M}_2 \boldsymbol{\Lambda}^* $$
		El producto interno se escribe: $\langle p,q \rangle_{\mathbf{M}_S} = \vec{p}^t \mathbf{M}_S \overline{\vec{q}}$.
	\end{alertblock}
\end{frame}

% --- DIAPOSITIVA 9B: MARCO TEÓRICO - PRODUCTOS INTERNOS DE SOBOLEV (IIB) ---
\begin{frame}{Marco Teórico: Productos Sobolev (IIb) - Matriz $\boldsymbol{\Lambda}$ y Casos Discretos}
	\begin{block}{Matriz de Derivación $\boldsymbol{\Lambda}$} \small
		$\boldsymbol{\Lambda}$ es la matriz del operador derivación $p(z) \mapsto p'(z)$ en la base monomial:
		\(
		\boldsymbol{\Lambda} = \begin{pmatrix}
			0      & 0      & 0      & \cdots \\
			1      & 0      & 0      & \cdots \\
			0      & 2      & 0      & \cdots \\
			\vdots & \vdots & \ddots & \ddots
		\end{pmatrix}
		\quad
		(\text{i.e., } (z^k)' = k z^{k-1})
		\)
	\end{block}
	\pause
	\begin{alertblock}{Caso Discreto} \small
		\begin{itemize}
			\item Si $\mu_2 = \delta_a$, su matriz de momentos $M(a)$ tiene entradas $(a^n \bar{a}^m)$.
		\end{itemize}
	\end{alertblock}
\end{frame}

% --- DIAPOSITIVA 11A.1: MARCO TEÓRICO - ESTADO DEL ARTE (IA) - ACOTACIÓN DE CEROS ---
\begin{frame}{Marco Teórico: Estado del Arte (Ia) - Acotación de Ceros}
	\begin{block}{Acotación del Operador Implica Acotación de Ceros} \small
		\textbf{Proposición \cite{escribano2025boundedness}:} Si $\mathcal{D}$ es acotado en $(\mathbb{P}[z], \langle\cdot,\cdot\rangle_M)$, entonces el conjunto de ceros $Z$ de los polinomios ortonormales está acotado. Además:
		$$ Z \subset \{ z \in \mathbb{C} : |z| < \|\mathcal{D}\| \} $$
	\end{block}
\end{frame}

% --- DIAPOSITIVA 11A.2: MARCO TEÓRICO - ESTADO DEL ARTE (IA) - DOMINACIÓN SECUENCIAL ---
\begin{frame}{Marco Teórico: Estado del Arte (Ia) - Dominación Secuencial}
	\begin{alertblock}{Dominación Secuencial (Definiciones)} \scriptsize
		(López Lagomasino, Pijeira \cite{lagomasino1999zero}; Escribano, Gonzalo \cite{escribano2025boundedness})
		\begin{itemize}
			\item \textbf{Medidas:} $\operatorname{supp}(\mu_2) \subset \operatorname{supp}(\mu_1)$ y $d\mu_2 = f_1 d\mu_1$ con $f_1 \in L_\infty(\mu_1)$.
			      \begin{itemize}
				      \item Caso real: $[c,d] \subset [a,b]$ y $w_2(t) \le C w_1(t)$.
			      \end{itemize}
			\item \textbf{Matrices:} $\{\mathbf{M}_1, \mathbf{M}_2\}$ es secuencialmente dominado si $\mathbf{M}_2 \le C \mathbf{M}_1$.
			      \begin{itemize}
				      \item Caso circunferencias: $S_{r_a}(a) \subset \mathbb{D}_{r_b}(b)$ (disco abierto).
			      \end{itemize}
		\end{itemize}
	\end{alertblock}
\end{frame}

% --- DIAPOSITIVA 11B: MARCO TEÓRICO - ESTADO DEL ARTE (IB) ---
\begin{frame}{Marco Teórico: Estado del Arte (Ib) - Dominación Secuencial (Teorema)}
	\begin{alertblock}{Dominación Secuencial (Teorema y Corolario)} \scriptsize
		(Escribano, Gonzalo \cite{escribano2025boundedness})
		\begin{itemize}
			\item \textbf{Teorema:} Si $\{\mathbf{M}_1, \mathbf{M}_2\}$ es secuencialmente dominado y los operadores $\mathcal{D}_j$ (para $\mathbf{M}_j$) son acotados, entonces $\mathcal{D}$ (para $\mathbf{M}_S$) es acotado.
			      \begin{itemize}
				      \item \textbf{Corolario:} Para $\langle p,q \rangle_S = \int_{S_{r_b}(b)} p\bar{q} + \int_{S_{r_a}(a)} p'\bar{q}'$, si $S_{r_a}(a) \subset \mathbb{D}_{r_b}(b)$, entonces $\mathcal{D}$ es acotado y $Z$ es acotado.
			      \end{itemize}
		\end{itemize}
	\end{alertblock}
	\pause \footnotesize
	\textit{Nota:} Hay casos donde $\mathcal{D}$ es acotado incluso sin dominación secuencial (e.g., $a \in \mathbb{D}_{r_b}(b)$ para el producto en circunferencias \cite{escribano2025zeros}).
\end{frame}

% --- DIAPOSITIVA 12A.1: MARCO TEÓRICO - ESTADO DEL ARTE (IIA) - PUNTOS DE EVALUACIÓN ACOTADOS (DEFINICIÓN) ---
\begin{frame}{Marco Teórico: Estado del Arte (IIa) - Puntos de Evaluación Acotados}
	Para productos con términos discretos como $|p'(a)|^2$.
	\begin{block}{Punto de Evaluación Acotado (bpe) \cite{escribano2025zeros}} \small
		$a \in \mathbb{C}$ es \textbf{bpe} de una medida $\mu$ (o matriz HPD $M$) si $\exists C>0$:
		$$ |p(a)|^2 \le C \int |p(z)|^2 d\mu(z) \quad (\text{o } |p(a)|^2 \le C \|p\|_M^2) \quad \forall p \in \mathbb{P}[z] $$
	\end{block}
\end{frame}

% --- DIAPOSITIVA 12A.2: MARCO TEÓRICO - ESTADO DEL ARTE (IIA) - BPE (PROPIEDADES Y LEMA) ---
\begin{frame}{Marco Teórico: Estado del Arte (IIa) - bpe (Propiedades y Lema)}
	\scriptsize
	\begin{itemize}
		\item \textbf{Envoltura convexa polinómica $P_C(K)$ \cite{escribano2025zeros}:} $\{z : |p(z)| \le \max_{K} |p(\xi)|, \forall p \in \mathbb{P}[z]\}$.
		      \begin{itemize}
			      \item Si $K$ es unión de intervalos reales, $P_C(K)=K$ \cite{Conway1978, Gamelin1969}.
			      \item Si $K=S_r(c)$, $P_C(K) = \overline{\mathbb{D}_r(c)}$ (disco cerrado) \cite{Conway1978, Gamelin1969}.
		      \end{itemize}
		\item \textbf{Lema \cite{escribano2025zeros}:} Todo bpe de $\mu$ está en $P_C(\operatorname{supp}(\mu))$.
		\item \textbf{Proposición \cite{escribano2025zeros}:} $a$ es bpe de $M \iff \{M, M(a)\}$ es secuencialmente dominado. ($M(a)$ es la matriz de momentos de $\delta_a$).
	\end{itemize}
\end{frame}

% --- DIAPOSITIVA 12B: MARCO TEÓRICO - ESTADO DEL ARTE (IIB) ---
\begin{frame}{Marco Teórico: Estado del Arte (IIb) - bpe (Resultado Clave)}
	\begin{alertblock}{Resultado Clave para Términos Discretos} \small
		\textbf{Proposición (Escribano, Gonzalo \cite{escribano2025zeros}):} Si $\mathcal{D}$ es acotado en $(\mathbb{P}[z], \|\cdot\|_M)$ y $a$ es bpe de $M$, entonces $\mathcal{D}$ es acotado también para la norma $\|p\|_S^2 = \|p\|_M^2 + |p'(a)|^2$.
		\begin{itemize}
			\item \textbf{Corolario \cite{escribano2025zeros}:} Para $\langle p,q \rangle_S = \int_{S_{r_b}(b)} p\bar{q} dm_{b;r_b} + p'(a)\overline{q'(a)}$, si $a \in \mathbb{D}_{r_b}(b)$ (disco abierto), entonces $\mathcal{D}$ es acotado y $Z$ es acotado.
		\end{itemize}
	\end{alertblock}
\end{frame}

% --- DIAPOSITIVA 13A.1: MARCO TEÓRICO - ESTADO DEL ARTE (IIIA) - OTROS ENFOQUES: SOPORTES DISJUNTOS ---
\begin{frame}{Marco Teórico: Estado del Arte (IIIa) - Otros Enfoques: Soportes Disjuntos}
	Existen enfoques que no dependen directamente de la acotación del operador $\mathcal{D}$.
	\begin{block}{Soportes Disjuntos (Caso Real Continuo)} \small
		\textbf{Teorema (López Lagomasino et al. \cite{lagomasino2001sobolev}):}
		Para $\langle p,q \rangle = \int_{\operatorname{supp}(\mu_1)} pq w_1 dt + \int_{\operatorname{supp}(\mu_2)} p'q' w_2 dt$.
		Si $\operatorname{co}(\operatorname{supp}(\mu_1)) \cap \operatorname{co}(\operatorname{supp}(\mu_2)) = \emptyset$:
		\begin{itemize}
			\item Los ceros de $\varphi_n$ están contenidos en un disco específico.
			\item El radio del disco es el diámetro de $\operatorname{co}(\operatorname{supp}(\mu_1)) \cup \operatorname{co}(\operatorname{supp}(\mu_2))$.
			\item El centro del disco es el punto de $\operatorname{co}(\operatorname{supp}(\mu_2))$ más alejado de $\operatorname{co}(\operatorname{supp}(\mu_1))$.
		\end{itemize}
		($\operatorname{co}(K)$ es la envoltura convexa de $K$).
	\end{block}
\end{frame}

% --- DIAPOSITIVA 13A.2: MARCO TEÓRICO - ESTADO DEL ARTE (IIIA) - SOPORTES DISJUNTOS (TEOREMA DETALLADO) ---
\begin{frame}{Marco Teórico: Estado del Arte (IIIa) - Soportes Disjuntos (Teorema Detallado)}
	\begin{block}{Soportes Disjuntos (Caso Real Continuo) - Continuación} \small
		Continuando con el teorema de López Lagomasino et al. para $\operatorname{co}(\operatorname{supp}(\mu_1)) \cap \operatorname{co}(\operatorname{supp}(\mu_2)) = \emptyset$:
		\begin{itemize}
			\item El radio del disco es el diámetro de $\operatorname{co}(\operatorname{supp}(\mu_1)) \cup \operatorname{co}(\operatorname{supp}(\mu_2))$.
			\item El centro del disco es el punto de $\operatorname{co}(\operatorname{supp}(\mu_2))$ más alejado de $\operatorname{co}(\operatorname{supp}(\mu_1))$.
		\end{itemize}
		($\operatorname{co}(K)$ es la envoltura convexa de $K$).
	\end{block}
\end{frame}

	\end{alertblock}
\end{frame}

% --- DIAPOSITIVA 21A.1: ANÁLISIS Y CONJETURAS (IA) - OBSERVACIONES (SEC. DOMINADO) ---
\begin{frame}{Análisis y Conjeturas (Ia): Observaciones (Sec. Dominado)}
	\begin{block}{Caso Secuencialmente Dominado (Continuo Real)} \small
		Si $I_2=[c,d] \subset I_1=[a,b]$ y $w_1(t)/w_0(t)$ acotado en $I_1$:
		\begin{itemize}
			\item El operador multiplicación $\mathcal{D}$ es acotado.
			\item $\|\mathcal{D}\| \le \sqrt{2(R^2 + \alpha)}$, donde $R = \max\{|a|,|b|,|c|,|d|\}$, $\alpha = \max_{t\in[a,b]}\frac{w_1(t)}{w_0(t)}$.
			\item Los ceros están acotados por $\|\mathcal{D}\|$.
		\end{itemize}
	\end{block}
\end{frame}

% --- DIAPOSITIVA 21A.2: ANÁLISIS Y CONJETURAS (IA) - OBSERVACIONES (VERIFICACIÓN EN EJEMPLOS) ---
\begin{frame}{Análisis y Conjeturas (Ia): Observaciones (Verificación en Ejemplos)}
	\begin{block}{Verificación en Ejemplos} \small
		Esto se verifica en ejemplos como el Ej.1 y Ej.2 del caso real.
	\end{block}
\end{frame}

% --- DIAPOSITIVA 21B: ANÁLISIS Y CONJETURAS (IB) - OBSERVACIONES CLAVE ---
\begin{frame}{Análisis y Conjeturas (Ib): Observaciones (Casos Generales)}
	\begin{alertblock}{Observaciones en Casos Generales (Ejemplos Numéricos)} \small
		En la totalidad de los ejemplos numéricos (reales y circunferencia):
		\begin{itemize}
			\item Incluso si $\mathcal{D}$ \textbf{no} es acotado (i.e., $\sigma_{1,n} \to \infty$):
			      \begin{itemize}
				      \item El radio espectral $\rho(\mathbf{D}_n)$ parece \textbf{permanecer acotado}.
				      \item El segundo valor singular $\sigma_{2,n}$ parece \textbf{converger} a $R = \max \{ \vert z \vert : z\in \operatorname{supp}(\mu_1)\cup \operatorname{supp}(\mu_2)\}$ (o $R_0 = \max \{ \vert z \vert : z\in \operatorname{supp}(\mu_1)\}$ si hay un átomo).
			      \end{itemize}
		\end{itemize}
	\end{alertblock}
\end{frame}

% --- DIAPOSITIVA 22A.1: CONJETURAS PRINCIPALES (IIA) - CONJETURA 1 (ACOTACIÓN) ---
\begin{frame}{Análisis y Conjeturas (IIa): Conjetura 1 (Acotación del Radio Espectral)}
	Basados en la evidencia numérica extensiva:

	\begin{block}{Conjetura 1: Acotación del Radio Espectral} \small
		El radio espectral de las matrices de multiplicación truncadas $\mathbf{D}_n$ permanece acotado incluso cuando el operador $\mathcal{D}$ no es acotado:
		$$ \lim_{n \to \infty} \rho(\mathbf{D}_n) < \infty $$
	\end{block}
\end{frame}

% --- DIAPOSITIVA 22A.2: CONJETURAS PRINCIPALES (IIA) - CONJETURA 1 (ASINTÓTICA) ---
\begin{frame}{Análisis y Conjeturas (IIa): Conjetura 1 (Comportamiento Asintótico)}
	\begin{block}{Conjetura 1: Acotación del Radio Espectral (Continuación)} \small
		Más aún, $\rho(\mathbf{D}_n)$ tiende a $R+k$, donde $k \in \mathbb{R}$ depende de la geometría de los soportes.
	\end{block}
\end{frame}

% --- DIAPOSITIVA 22b: CONJETURAS PRINCIPALES (IIb) - FORMULACIÓN ---
\begin{frame}{Análisis y Conjeturas (IIb): Conjetura 2 y Soporte Teórico}
	\begin{block}{Conjetura 2: Convergencia del Segundo Valor Singular} \small
		El segundo mayor valor singular $\sigma_{2,n}$ de $\mathbf{D}_n$ converge:
		$$ \lim_{n \to \infty} \sigma_{2,n} < \infty $$
		Específicamente, converge a $R = \max \{ \vert z \vert : z\in \operatorname{supp}(\mu_1)\cup \operatorname{supp}(\mu_2)\}$ en casos continuos, o a $R_0 = \max \{ \vert z \vert : z\in \operatorname{supp}(\mu_1)\}$ en casos con un átomo para la derivada.
	\end{block}
	\pause
	\begin{exampleblock}{Soporte Teórico} \small
		Resultados parciales y análisis teóricos (e.g., \cite{escribano2025boundedness, singularValuesOfHessenbergMatrices}) apoyan estas conjeturas, utilizando herramientas como el teorema de entrelazamiento de Cauchy y propiedades de matrices de Hessenberg.
	\end{exampleblock}
\end{frame}

% --- DIAPOSITIVA 23A.1: ANÁLISIS Y CONJETURAS (IIIA) - TRASLACIONES (OBSERVACIÓN Y CONJETURA) ---
\begin{frame}{Análisis y Conjeturas (IIIa): Traslaciones de Soportes (Observación y Conjetura)}
	\begin{block}{Observación Empírica sobre Traslación de Soportes} \small
		\begin{itemize}
			\item Cuando los soportes (en $\mathbb{R}$) se trasladan a lo largo del eje real, los ceros parecen trasladarse de igual manera.
			\item \textbf{Conjetura:} Los ceros de los polinomios de Sobolev asociados a medidas trasladadas en el eje real, se trasladan mediante la misma traslación.
		\end{itemize}
	\end{block}
\end{frame}

% --- DIAPOSITIVA 23A.2: ANÁLISIS Y CONJETURAS (IIIA) - TRASLACIONES (LIMITACIONES) ---
\begin{frame}{Análisis y Conjeturas (IIIa): Traslaciones de Soportes (Limitaciones)}
	\begin{block}{Observación Empírica sobre Traslación de Soportes (Continuación)} \small
		\begin{itemize}
			\item[\textbullet] Sin embargo, si la traslación tiene componente imaginaria, se observan comportamientos anómalos.
		\end{itemize}
	\end{block}
\end{frame}

% --- DIAPOSITIVA 23B: ANÁLISIS Y CONJETURAS (IIIB) - SEMEJANZA ---
\begin{frame}{Análisis y Conjeturas (IIIb): Transformaciones de Semejanza}
	\begin{alertblock}{Transformaciones de Semejanza} \small
		\begin{itemize}
			\item En el caso estándar (una medida), si $f(z)=\alpha z+\beta$ es una semejanza, los ceros de $P_n(z)$ se transforman a los ceros de $P_n^f(z)$ mediante $f$.
			\item \textbf{Precaución:} En productos de Sobolev, la aplicabilidad directa de este resultado \textbf{no es general}, como muestran ejemplos con traslaciones complejas. Se requiere más investigación.
		\end{itemize}
	\end{alertblock}
\end{frame}
